        let currentLang = document.body.getAttribute('dir') === 'rtl' ? 'ar' : 'en';

        function applyTranslations(lang) {
            currentLang = lang;
            document.body.setAttribute('dir', lang === 'ar' ? 'rtl' : 'ltr');
            
            // تحديث زر اللغة النشط
            document.querySelectorAll('.language-btn').forEach(btn => {
                btn.classList.toggle('active', btn.dataset.lang === lang);
            });

            // تطبيق الترجمات
            document.querySelectorAll('[data-i18n]').forEach(element => {
                const key = element.getAttribute('data-i18n');
                if (translations[lang] && translations[lang][key]) {
                    let text = translations[lang][key];

                    if (key === "cart.itemsTitle" || key === "cart.summary.subtotal") {

            // بما أن العناوين تعتمد على الترجمة، نعيد تحديث الإجمالي أيضاً
            updateCartTotal();

        // =========================================================
        // 2. Cart Interaction Logic (الجزء المضاف للتحسين)
        // =========================================================
        
        const SHIPPING_COST = 600; // تكلفة الشحن بالدينار الجزائري (DA)

        /**
         * دالة مساعدة لتحويل الرقم إلى تنسيق نصي (4,500 DA أو 4,500 دج)
         * @param {number} priceNumber - السعر كرقم.
         * @returns {string} - السعر بتنسيق نصي.
         */
        function formatPrice(priceNumber) {
            const currencySymbol = currentLang === 'ar' ? ' دج' : ' DA';
            // استخدام توطين (locale) لإضافة الفواصل
            return priceNumber.toLocaleString('en-US', {
                minimumFractionDigits: 0,
                maximumFractionDigits: 0
            }) + currencySymbol;
        }

        /**
         * الدالة الرئيسية: لإعادة حساب إجمالي السلة بالكامل
         */
        function updateCartTotal() {
            let subtotal = 0;
            let totalUniqueItems = 0;
            const cartItems = document.querySelectorAll('.cart-item');

            // 1. حساب الإجمالي الفرعي (Subtotal)
            cartItems.forEach(card => {
                const unitPrice = parseFloat(card.dataset.unitPrice || 0);
                const quantityElement = card.querySelector('.quantity');
                const quantity = parseInt(quantityElement.textContent);
                
                // حساب سعر المنتج (الكمية * سعر الوحدة)
                const productTotalPrice = unitPrice * quantity;

                // تحديث السعر الإجمالي للمنتج الواحد في الواجهة
                card.querySelector('.item-price').textContent = formatPrice(productTotalPrice);

                subtotal += productTotalPrice;
                totalUniqueItems++;
            });

            // 2. حساب الإجمالي النهائي (Total)
            const total = subtotal + SHIPPING_COST;
            
            // 3. تحديث ملخص الطلب (Summary)
            document.getElementById('subtotal-price').textContent = formatPrice(subtotal);
            document.getElementById('shipping-price').textContent = formatPrice(SHIPPING_COST);
            document.getElementById('total-price').textContent = formatPrice(total);
            
            // 4. تحديث عدد المنتجات في الترجمة ورأس السلة
            const cartItemsTitle = document.querySelector('.cart-items h2');
            const subtotalTitleElement = document.querySelector('[data-i18n="cart.summary.subtotal"]');
            const cartCountHeader = document.querySelector('.cart-count');

            if (cartItemsTitle) {
                cartItemsTitle.textContent = translations[currentLang]["cart.itemsTitle"].replace('{{count}}', totalUniqueItems);
            }
            if (subtotalTitleElement) {
                subtotalTitleElement.textContent = translations[currentLang]["cart.summary.subtotal"].replace('{{count}}', totalUniqueItems);
            }
            if (cartCountHeader) {
                cartCountHeader.textContent = totalUniqueItems;
            }
            
            // 5. إظهار/إخفاء حالة السلة الفارغة
            const emptyCartElement = document.querySelector('.empty-cart');
            const cartLayout = document.querySelector('.cart-layout');
            
            if (totalUniqueItems === 0) {
                 if (emptyCartElement) {
                    emptyCartElement.style.display = 'block';
                 }
                 if (cartLayout) {
                    cartLayout.style.display = 'none';
                 }
            } else {
                 if (emptyCartElement) {
                    emptyCartElement.style.display = 'none';
                 }
                 if (cartLayout) {
                    // نستخدم Flex لأن هذا ما تم تعريفه في CSS
                    cartLayout.style.display = 'flex'; 
                 }
            }
        }

        /**
         * إعداد مستمعي الأحداث لأزرار الكمية والحذف
         */
        function setupCartInteractions() {
            const cartSection = document.querySelector('.cart-section');

            // استخدام تفويض الحدث (Event Delegation)
            cartSection.addEventListener('click', function(event) {
                const target = event.target;
                const card = target.closest('.cart-item');

                // إذا لم يكن الضغط داخل عنصر المنتج، نخرج
                if (!card) return; 

                const quantityElement = card.querySelector('.quantity');
                let quantity = parseInt(quantityElement.textContent);

                // 1. زيادة/نقصان الكمية
                if (target.classList.contains('plus') || target.closest('.plus')) {
                    quantity++;
                    quantityElement.textContent = quantity;
                    updateCartTotal(); 
                } 
                else if (target.classList.contains('minus') || target.closest('.minus')) {
                    if (quantity > 1) { // لا يمكن أن تقل الكمية عن 1
                        quantity--;
                        quantityElement.textContent = quantity;
                        updateCartTotal();
                    }
                } 
                
                // 2. حذف المنتج نهائياً
                else if (target.classList.contains('remove-btn') || target.closest('.remove-btn')) {
                    // رسالة تأكيد (اختياري)
                    if (confirm(currentLang === 'ar' ? "هل أنت متأكد من رغبتك في حذف هذا المنتج من السلة؟" : "Are you sure you want to remove this item from the cart?")) {
                        card.remove(); 
                        updateCartTotal(); 
                    }
                }
            });
        }

            applyTranslations(currentLang);

            setupCartInteractions();
            
            // التأكد من تحديث الإجمالي بناءً على القيم الافتراضية
            updateCartTotal(); 
        });
