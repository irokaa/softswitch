// متغيرات النطاق العام (Global Scope)
let currentLanguage = 'en';
let currentProductKey = null;

// تعاريف العناصر (يجب أن تكون خارج أي دالة لعدم التكرار)
const body = document.body;
const dropdownContainer = document.querySelector('.language-dropdown');
const toggleButton = document.getElementById('lang-toggle-button');
const dropdownMenu = document.getElementById('lang-dropdown-menu');
const currentLangText = document.querySelector('.current-lang-text');
// **ملاحظة: يجب تعريف dropdownItems بعد التحقق من وجود dropdownMenu**
let dropdownItems = [];
if (dropdownMenu) {
    dropdownItems = dropdownMenu.querySelectorAll('li');
}
const translatableElements = document.querySelectorAll('.translatable');

// 💡 الوظيفة الرئيسية لتغيير اللغة (المسماة setLanguage في الكود السابق)
function changeLanguage(lang, langText) {
    // تحديث متغير اللغة الحالية
    currentLanguage = lang;

    // أ. تغيير اتجاه الصفحة (dir)
    if (lang === 'ar') {
        body.setAttribute('dir', 'rtl');
    } else {
        body.setAttribute('dir', 'ltr');
    }

    // ب. تحديث المحتوى النصي للعناصر المترجمة
    translatableElements.forEach(element => {
        const translation = element.getAttribute(`data-${lang}`);
        if (translation) {
            element.textContent = translation;
        }
    });

    // ج. تحديث حالة القائمة المنسدلة (النص النشط)
    if (currentLangText) {
        currentLangText.textContent = langText;
    }
    if (dropdownItems.length > 0) {
        dropdownItems.forEach(item => item.classList.remove('active'));
        const activeItem = document.querySelector(`.dropdown-menu li[data-lang="${lang}"]`);
        if (activeItem) {
            activeItem.classList.add('active');
        }
    }
    
    // د. إغلاق القائمة المنسدلة (إذا كانت مفتوحة)
    if (dropdownContainer) {
        dropdownContainer.classList.remove('open');
        toggleButton.setAttribute('aria-expanded', 'false');
    }

    // هـ. حفظ اللغة المختارة
    localStorage.setItem('selectedLang', lang);
    localStorage.setItem('selectedLangText', langText);
    
    // **مهم:** تحديث تفاصيل المنتج إذا كانت معروضة حاليًا
    if (currentProductKey) {
        updateProductDetails(currentProductKey, currentLanguage);
    }
}


// 🚀 وظيفة تهيئة مستمعي الأحداث (تُستدعى مرة واحدة فقط عند تحميل الصفحة)
function initializeListeners() {
    // مستمعي القائمة المنسدلة للغة
    if (dropdownContainer && toggleButton && dropdownMenu) {
        // 1. وظيفة لفتح/إغلاق القائمة المنسدلة
        toggleButton.addEventListener('click', (e) => {
            dropdownContainer.classList.toggle('open');
            const isExpanded = dropdownContainer.classList.contains('open');
            toggleButton.setAttribute('aria-expanded', isExpanded);
            e.stopPropagation(); 
        });

        // إغلاق القائمة عند النقر خارجها
        document.addEventListener('click', (e) => {
            if (!dropdownContainer.contains(e.target)) {
                dropdownContainer.classList.remove('open');
                toggleButton.setAttribute('aria-expanded', 'false');
            }
        });

        // 3. إضافة مستمعي الأحداث لعناصر القائمة
        dropdownItems.forEach(item => {
            item.addEventListener('click', () => {
                const lang = item.getAttribute('data-lang');
                const langText = item.textContent;
                // **استدعاء changeLanguage هنا**
                changeLanguage(lang, langText); 
            });
        });
    }

    // إضافة مستمعي الأحداث لأزرار "عرض التفاصيل"
    document.querySelectorAll('.view-details').forEach(button => {
        button.addEventListener('click', (event) => {
            const productKey = event.currentTarget.getAttribute('data-product');
            showProductDetails(productKey);
        });
    });
    
    // ... أضف هنا مستمعي أحداث تصفية الفئات، و toggleMenu، إلخ.
    // (لعدم تكرار الكود الموجود في الأسفل)
}
